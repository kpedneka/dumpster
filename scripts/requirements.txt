# Dependencies for scripts/extract_entities.py, the local entity-extraction
# sidecar invoked by internal/entity/gliner via subprocess.
#
# `pip install` alone does NOT install en_core_web_sm — that's a separate
# model download this project previously missed everywhere (local dev and
# the Docker image alike), silently degrading sentence segmentation. After
# installing these, also run:
#   python -m spacy download en_core_web_sm
#
# GLiNER.from_pretrained() also checks HuggingFace Hub for updates on every
# call by default, even once its weights are cached. The runtime image bakes
# the model in and forces offline mode (see Dockerfile) so this never runs
# there at all. For local dev, after the model has been fetched once, set
# these to make it fully offline too:
#   export HF_HUB_OFFLINE=1 TRANSFORMERS_OFFLINE=1
spacy>=3.7,<4
gliner>=0.2,<1

# Dependencies for scripts/extract_regions.py, the PDF region classifier
# invoked by internal/manifest/layout via subprocess (no VLM layer here —
# see RegionClassificationHandler's type doc for why). pymupdf is
# AGPL-3.0/commercial dual-licensed (see LICENSE at the repo root).
# unstructured[pdf] dropped: it used to cover figure detection and a
# table-extraction fallback, but shared the same silent-failure mode on
# malformed PDFs that motivated dropping it for text too — see
# extract_regions()'s docstring for the full reasoning.
pymupdf>=1.24,<2
pdfplumber>=0.11,<1
Pillow>=10,<12

# Dependencies for scripts/inference_service.py (v4.8): the consolidated,
# always-on HTTP service that wraps extract_entities.py/extract_regions.py's
# pipeline logic plus local embeddings behind /entities, /regions, and
# /embeddings endpoints, replacing the per-Go-process subprocess sidecar
# pattern. fastapi+uvicorn are new dependencies purely for this service —
# extract_entities.py/extract_regions.py keep working standalone via their
# existing stdin/stdout protocol until v4.9/v4.10 retire those call sites.
fastapi>=0.115,<1
uvicorn>=0.32,<1
sentence-transformers>=3.2,<4
httpx>=0.27,<1
